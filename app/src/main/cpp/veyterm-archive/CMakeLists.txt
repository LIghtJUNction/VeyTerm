cmake_minimum_required(VERSION 3.18)
project(veyterm_archive C)
find_package(JNI REQUIRED)
find_package(ZLIB REQUIRED)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/sources.txt" ARCHIVE_NAMES)
set(ARCHIVE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/source" CACHE PATH "Prepared libarchive sources")
set(ARCHIVE_SOURCES)
foreach(name IN LISTS ARCHIVE_NAMES)
  list(APPEND ARCHIVE_SOURCES "${ARCHIVE_SOURCE_DIR}/${name}")
endforeach()
add_library(rar_reader OBJECT ${ARCHIVE_SOURCES})
target_compile_definitions(rar_reader PRIVATE VEYTERM_ARCHIVE_BUDGET HAVE_CONFIG_H _GNU_SOURCE)
add_library(veyterm_archive SHARED rar_jni.c rar_alloc.c $<TARGET_OBJECTS:rar_reader>)
foreach(target rar_reader veyterm_archive)
  target_include_directories(${target} PRIVATE . "${ARCHIVE_SOURCE_DIR}" ${JNI_INCLUDE_DIRS})
  target_compile_options(${target} PRIVATE -std=c11 -Wall -Wextra -fvisibility=hidden -ffunction-sections -fdata-sections -fstack-protector-strong)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
target_compile_definitions(veyterm_archive PRIVATE _GNU_SOURCE)
target_link_libraries(veyterm_archive PRIVATE ZLIB::ZLIB)
target_link_options(veyterm_archive PRIVATE -Wl,--no-undefined -Wl,--gc-sections "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
